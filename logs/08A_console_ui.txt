"""Reusable ANSI console styling utilities."""

import os
import sys


COLORS = {
    "reset": "\033[0m",
    "bold": "\033[1m",
    "gray": "\033[90m",
    "red": "\033[91m",
    "green": "\033[92m",
    "yellow": "\033[93m",
    "blue": "\033[94m",
    "purple": "\033[95m",
    "cyan": "\033[96m",
    "white": "\033[97m",
}

WIDTH = 66


def color_enabled():
    """Disable ANSI colors for redirected output."""

    if "NO_COLOR" in os.environ:
        return False

    if os.environ.get("TERM") == "dumb":
        return False

    return bool(sys.stdout.isatty())


def paint(value, color="white", bold=False):
    """Apply a color and restore the default style."""

    text = str(value)

    if not color_enabled():
        return text

    prefix = COLORS[color]

    if bold:
        prefix += COLORS["bold"]

    return prefix + text + COLORS["reset"]


def line(character="-"):
    """Print a neutral separator."""

    print(paint(character * WIDTH, "gray"))


def header(title, subtitle=None, color="yellow"):
    """Display a section title with vertical spacing."""

    print()
    line("=")

    print(paint(title, color, bold=True))

    if subtitle:
        print(paint(subtitle, "gray"))

    line("=")
    print()


def field(label, value, color="white"):
    """Display a labeled value."""

    print(
        f" {label} : "
        f"{paint(value, color, bold=True)}"
    )


def menu_item(number, label):
    """Display a numbered menu item."""

    print(
        f" {paint(f'[{number}]', 'cyan', True)} "
        f" {label}"
    )


def prompt(message):
    """Read user input using a colored prompt."""

    return input(paint(message, "green", True))


def message(text, color="white"):
    """Display a separated status message."""

    print()
    print(paint(text, color, True))
    print()


def progress_bar(current, total, width=30):
    """Display a text-based progress bar."""

    ratio = current / total if total else 0
    ratio = max(0.0, min(1.0, ratio))

    filled = int(ratio * width)

    bar = (
        paint("#" * filled, "cyan")
        + paint("-" * (width - filled), "gray")
    )

    print(f" [{bar}] {ratio * 100:.0f}%")


def score_bar(score):
    """Display a normalized score bar."""

    progress_bar(score, 100)

def read_text(prompt):

    while True:

        value = ui.prompt(prompt).strip()

        if value:
            return value

        ui.message(
            "빈칸은 입력할 수 없습니다.",
            "red",

def read_int(prompt, minimum, maximum):

    while True:

        value = ui.prompt(prompt).strip()

        try:
            number = int(value)

        except ValueError:

            ui.message(
                "숫자를 입력하세요.",
                "red",
            )

            continue

        if minimum <= number <= maximum:
            return number

        ui.message(
            f"{minimum}~{maximum} 사이의 "
            "숫자를 입력하세요.",

        )

        count=count,
        random_order=True,
        shuffle_choices=True,
    )

def add_question(game, storage=None):

    ui.header("ADD QUESTION")

    question = read_text("문제 내용: ")

    if any(

    ):

        ui.message(
            "이미 등록된 문제입니다.",
            "yellow",

        return

    print()

                )

            choices.append(choice)
            break

    print()

    answer = read_int(

    level = read_int(

        1,
        2,
    )

    difficulty = (

        if level == 1
        else "intermediate"
    )

    print()

def remove_question(game, storage=None):

    quizzes = game.list_quizzes()

    if not quizzes:

        ui.message(
            "삭제할 문제가 없습니다.",
            "yellow",
        )

        return game

    ui.header("DELETE QUESTION", color="red")

    list_questions(game)

    valid_ids = {
        quiz.id
        for quiz in quizzes
    }

    while True:

        quiz_id = read_int(
            "삭제할 문제 ID (0=취소): ",
            0,
            max(valid_ids),
        )

        if quiz_id == 0:

            ui.message(
                "삭제를 취소했습니다.",
                "yellow",
            )

            return game

        if quiz_id in valid_ids:
            break

        ui.message(
            "존재하지 않는 문제 ID입니다.",
            "red",
        )

    selected = next(
        quiz
        for quiz in quizzes
        if quiz.id == quiz_id
    )

    ui.message(
        f"삭제 대상: {selected.question}",
        "red",
    )

    while True:

        answer = ui.prompt(
            "정말 삭제하시겠습니까? (Y/N): "
        ).strip().upper()

        if answer == "N":

            ui.message(
                "삭제를 취소했습니다.",
                "yellow",
            )

            return game

        if answer == "Y":
            break

        ui.message(
            "Y 또는 N을 입력하세요.",
            "red",
        )

    replacement = delete_question(
        game,
        quiz_id,
    )

    if not save_state(replacement, storage):

        ui.message(
            "저장 실패로 삭제를 취소합니다.",
            "red",
        )

        return game

    ui.message(
        f"문제 삭제 완료! ID: {quiz_id}",
        "green",
    )

    ui.field(
        "전체 등록 문제",
        f"{len(replacement.list_quizzes())}개",
        "cyan",
    )

    return replacement

def run_menu(game, storage=None):

    try:

        while True:

            show_menu(game)

            choice = read_int(
                "메뉴 선택 (0~6): ",
                0,
                6,
            )

            if choice == 0:

                if not save_state(game, storage):

                    ui.message(
                        "저장하지 못했습니다. "
                        "다시 종료를 시도하세요.",
                        "red",
                    )

                    continue

                ui.message(
                    "프로그램을 종료합니다.",
                    "cyan",
                )

                return 0

            try:

                if choice == 1:
                    play_quiz(game, storage)

                elif choice == 2:
                    add_question(game, storage)

                elif choice == 3:
                    list_questions(game)

                elif choice == 4:
                    show_score(game)

                elif choice == 5:

                    game = remove_question(
                        game,
                        storage,
                    )

                elif choice == 6:
                    show_history(game)

            except (ValueError, RuntimeError) as error:

                game.cancel_game()

                ui.message(
                    f"오류: {error}",
                    "red",
                )

    except (KeyboardInterrupt, EOFError):

        game.cancel_game()

        ui.message(
            "입력이 중단되었습니다.",
            "yellow",
        )

        if save_state(game, storage):

            ui.message(
                "프로그램을 안전하게 종료합니다.",
                "cyan",
            )

            return 0

        ui.message(
            "저장 실패 상태로 프로그램을 종료합니다.",
            "red",
        )

        return 1

            "console_ui.color_enabled",
            return_value=True,
        ):

            self.assertEqual(
                ui.paint("Correct", "blue"),
                "\033[94mCorrect\033[0m",

    def test_02_red_color(self):

        with patch(
            "console_ui.color_enabled",
            return_value=True,
        ):

            self.assertEqual(
                ui.paint("Incorrect", "red"),
                "\033[91mIncorrect\033[0m",
            )

    def test_03_plain_text(self):

        with patch(
            "console_ui.color_enabled",
            return_value=False,
        ):

            self.assertEqual(
                ui.paint("Plain text", "blue"),
                "Plain text",
            )

    def test_04_bold_text(self):

        with patch(
            "console_ui.color_enabled",
            return_value=True,
        ):

            self.assertIn(
                "\033[1m",
                ui.paint("Title", "cyan", True),
            )

    def test_05_no_color_environment(self):

        with patch.dict(
            os.environ,
            {"NO_COLOR": "1"},
        ):

            self.assertFalse(ui.color_enabled())

        output = io.StringIO()

        with redirect_stdout(output):

            ui.header("TEST")

        self.assertNotIn(
            "\033[",
            output.getvalue(),
        )

def timed(function, *args, **kwargs):

    start = time.perf_counter()

    result = function(
        *args,
        **kwargs
    )

    elapsed = (
        time.perf_counter() - start
    ) * 1000

    return result, elapsed


def create_quizzes(count, template):

    quizzes = []

    for index in range(1, count + 1):

        item = dict(template)

        item["id"] = index

        item["question"] = (
            f"Benchmark question {index:06d}"
        )

        item["choices"] = list(
            template["choices"]
        )

        quizzes.append(
            Quiz.from_dict(item)
        )

    return quizzes


def search_by_id(quizzes, target):

    return next(
        (
            quiz
            for quiz in quizzes
            if quiz.id == target
        ),
        None
    )


def measure_search(quizzes, target):

    measurements = []

    for _ in range(SEARCH_REPEATS):

        found, elapsed = timed(
            search_by_id,
            quizzes,
            target
        )

        assert found is not None
        assert found.id == target

        measurements.append(elapsed)

    return statistics.median(
        measurements
    )


def measure_object_memory(count, template):

    gc.collect()

    tracemalloc.start()

    try:

        quizzes = create_quizzes(
            count,
            template
        )

        assert len(quizzes) == count

        current, peak = (
            tracemalloc.get_traced_memory()
        )

        return peak / (1024 ** 2)

    finally:

        tracemalloc.stop()


def run_case(count, template):

    print(
        f"\nCASE: {count} QUESTIONS",
        flush=True
    )

    random.seed(SEED)

    gc.collect()

    quizzes, create_ms = timed(
        create_quizzes,
        count,
        template
    )

    assert len(quizzes) == count

    print(
        f"CREATE OBJECTS: {create_ms:.6f} ms",
        flush=True
    )

    game, init_ms = timed(
        QuizGame,
        quizzes
    )

    print(
        f"INITIALIZE GAME: {init_ms:.6f} ms",
        flush=True
    )

    listed, list_ms = timed(
        game.list_quizzes
    )

    assert len(listed) == count

    print(
        f"LIST QUESTIONS: {list_ms:.6f} ms",
        flush=True
    )

    search_ms = measure_search(
        listed,
        count
    )

    print(
        f"SEARCH LAST ID: {search_ms:.6f} ms",
        flush=True
    )

    _, random_ms = timed(
        game.start_game,
        count=SESSION_SIZE
    )

    assert game.is_active

    assert game.current_quiz is not None

    print(
        f"RANDOM SESSION: {random_ms:.6f} ms",
        flush=True
    )

    game.cancel_game()

    assert not game.is_active

    with tempfile.TemporaryDirectory() as directory:

        path = (
            Path(directory)
            / "state.json"
        )

        storage = QuizStorage(
            path=path
        )

        _, save_ms = timed(
            storage.save,
            game
        )

        assert path.is_file()

        size_kib = (
            path.stat().st_size / 1024
        )

        print(
            f"JSON SAVE: {save_ms:.6f} ms",
            flush=True
        )

        restored, load_ms = timed(
            storage.load
        )

        print(
            f"JSON LOAD: {load_ms:.6f} ms",
            flush=True
        )

        restored_quizzes = (
            restored.list_quizzes()
        )

        assert len(restored_quizzes) == count

        expected = [
            quiz.to_dict()
            for quiz in listed
        ]

        actual = [
            quiz.to_dict()
            for quiz in restored_quizzes
        ]

        assert actual == expected

        assert (
            restored.best_score
            == game.best_score
        )

        assert not path.with_name(
            "state.json.tmp"
        ).exists()

    memory_mib = measure_object_memory(
        count,
        template
    )

    print(
        f"OBJECT MEMORY PEAK: {memory_mib:.6f} MiB",
        flush=True
    )

    print(
        f"JSON FILE SIZE: {size_kib:.6f} KiB",
        flush=True
    )

    print(
        f"PASS: {count} question round trip",
        flush=True
    )

    return {
        "count": count,
        "create_ms": create_ms,
        "init_ms": init_ms,
        "list_ms": list_ms,
        "search_ms": search_ms,
        "random_ms": random_ms,
        "save_ms": save_ms,
        "load_ms": load_ms,
        "memory_mib": memory_mib,
        "size_kib": size_kib,
    }

def main():

    parser = argparse.ArgumentParser()

    parser.add_argument(
        "--smoke",
        action="store_true",
        help="Run only 20 questions without writing a report."
    )

    args = parser.parse_args()

    defaults = json.loads(
        DEFAULT_PATH.read_text(
            encoding="utf-8"
        )
    )

    assert len(defaults) == 20

    template = defaults[0]

    sizes = (
        (20,)
        if args.smoke
        else SIZES
    )

    start = time.perf_counter()

    results = [
        run_case(count, template)
        for count in sizes
    ]

    elapsed = (
        time.perf_counter() - start
    )

    if not args.smoke:

        write_document(
            results,
            elapsed
        )

        print(
            "CREATED:",
            "docs/SCALABILITY_ANALYSIS.md"
        )

    print("\nBENCHMARK SUMMARY")

    for row in results:

        print(
            "RESULT:",
            json.dumps(
                row,
                sort_keys=True
            )
        )

    print("CASES COMPLETED:", len(results))

    print(
        "TOTAL BENCHMARK SECONDS:",
        f"{elapsed:.3f}"
    )

    print(
        "PASS: All benchmark cases verified"
    )


if __name__ == "__main__":
    main()
